function doGet() { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dados"); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return ContentService .createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } const values = sheet.getRange(2, 1, lastRow - 1, 8).getValues(); const data = values .filter(row => row.some(cell => String(cell).trim() !== "")) .map(row => ({ id: row[0] || "", nome: row[1] || "", caminhao: row[2] || "", bomba: row[3] || "", volume: row[4] || "", data: row[5] || "", observacao: row[6] || "", timestamp: row[7] || "" })); return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dados"); const data = JSON.parse(e.postData.contents || "{}"); const action = String(data.action || "create").trim(); const id = String(data.id || "").trim(); if (action === "update") { if (!id) { throw new Error("ID inválido para atualização."); } const row = localizarLinhaPorId(sheet, id); if (!row) { throw new Error("Registro não encontrado para atualização."); } sheet.getRange(row, 1, 1, 8).setValues([[ id, data.nome || "", data.caminhao || "", data.bomba || "", data.volume || "", data.data || "", data.observacao || "", new Date() ]]); return ContentService .createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } if (action === "delete") { if (!id) { throw new Error("ID inválido para exclusão."); } const row = localizarLinhaPorId(sheet, id); if (!row) { throw new Error("Registro não encontrado para exclusão."); } sheet.deleteRow(row); return ContentService .createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } const newId = Utilities.getUuid(); sheet.appendRow([ newId, data.nome || "", data.caminhao || "", data.bomba || "", data.volume || "", data.data || "", data.observacao || "", new Date() ]); return ContentService .createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function localizarLinhaPorId(sheet, id) { const lastRow = sheet.getLastRow(); if (lastRow < 2) return null; const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (let i = 0; i < ids.length; i++) { if (String(ids[i][0]).trim() === id) { return i + 2; } } return null; }
