function doGet() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dados"); const values = sheet.getDataRange().getValues(); if (values.length <= 1) { return ContentService .createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } const rows = values.slice(1); const data = rows.map((row, index) => ({ rowIndex: index + 2, nome: row[0] || "", caminhao: row[1] || "", bomba: row[2] || "", volume: row[3] || "", data: row[4] || "", observacao: row[5] || "", timestamp: row[6] || "" })); return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dados"); const raw = e && e.postData && e.postData.contents ? e.postData.contents : "{}"; const body = JSON.parse(raw); const action = e && e.parameter && e.parameter.action ? String(e.parameter.action).trim() : "create"; const rowIndex = e && e.parameter && e.parameter.rowIndex ? Number(e.parameter.rowIndex) : null; if (action === "update") { if (!rowIndex || rowIndex < 2 || rowIndex > sheet.getLastRow()) { throw new Error("rowIndex inválido para atualização."); } sheet.getRange(rowIndex, 1, 1, 7).setValues([[ body.nome || "", body.caminhao || "", body.bomba || "", body.volume || "", body.data || "", body.observacao || "", new Date() ]]); return ContentService .createTextOutput(JSON.stringify({ success: true, action: "update", rowIndex: rowIndex })) .setMimeType(ContentService.MimeType.JSON); } if (action === "delete") { if (!rowIndex || rowIndex < 2 || rowIndex > sheet.getLastRow()) { throw new Error("rowIndex inválido para exclusão."); } sheet.deleteRow(rowIndex); return ContentService .createTextOutput(JSON.stringify({ success: true, action: "delete", rowIndex: rowIndex })) .setMimeType(ContentService.MimeType.JSON); } sheet.appendRow([ body.nome || "", body.caminhao || "", body.bomba || "", body.volume || "", body.data || "", body.observacao || "", new Date() ]); return ContentService .createTextOutput(JSON.stringify({ success: true, action: "create" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: error.message || String(error) })) .setMimeType(ContentService.MimeType.JSON); } }
